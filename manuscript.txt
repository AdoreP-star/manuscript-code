r6_read <- function(file) {
  d=read.csv(file,check.names=FALSE,na.strings=c("","NA","NaN","NULL"),
             strip.white=TRUE,fileEncoding="GB18030")
  if("HGB" %in% names(d) && !"HB" %in% names(d)) names(d)[names(d)=="HGB"]="HB"
  if(!all(c("Result",CANDIDATE_FEATURES) %in% names(d)))
    stop(file,": 缺候选变量 ",paste(setdiff(c("Result",CANDIDATE_FEATURES),names(d)),collapse=","))
  for(nm in c("Result",CANDIDATE_FEATURES)) {
    z=suppressWarnings(as.numeric(as.character(d[[nm]])))
    if(any(is.na(z) & !is.na(d[[nm]]))) stop(file,": 非数值内容：",nm)
    if(any(is.infinite(z))) stop(file,": 无穷值：",nm)
    d[[nm]]=z
  }
  if(anyNA(d$Result) || !all(d$Result %in% 0:1)) stop("结局缺失或不是0/1")
  for(nm in BINARY_FEATURES) if(!all(na.omit(d[[nm]]) %in% 0:1)) stop(nm,"不是0/1编码")
  d$.source_row=seq_len(nrow(d))
  d$Result=factor(d$Result,levels=0:1,labels=c("No","Yes"))
  d
}
r6_cohort <- function(file,cohort,id,stay,timefile=NULL) {
  d=r6_read(file)
  has_ids=all(c(id,stay) %in% names(d)) && !anyNA(d[[id]]) && !anyNA(d[[stay]])
  if(has_ids) {
    if(anyDuplicated(d[[stay]])) stop(cohort,": ICU记录ID重复")
    d$.patient=as.character(d[[id]]); d$.stay=as.character(d[[stay]])
    has_time=!is.null(timefile) && file.exists(timefile)
    if(has_time) {
      tt=read.csv(timefile,check.names=FALSE,stringsAsFactors=FALSE)
      if(!all(c(stay,"icu_intime") %in% names(tt)))
        stop(cohort,": 住院时间表缺少 ",stay," 或 icu_intime")
      ii=match(d$.stay,as.character(tt[[stay]]))
      if(anyNA(ii)) stop(cohort,": 住院时间表匹配失败")
      d$.admission=as.POSIXct(tt$icu_intime[ii],format="%d/%m/%Y %H:%M:%S",tz="UTC")
      if(anyNA(d$.admission)) stop(cohort,": 入ICU时间不能按日/月/年读取")
    } else {
      d$.admission=as.POSIXct(rep(0,nrow(d)),origin="1970-01-01",tz="UTC")
      warning(cohort,": 未提供可用住院时间表；同一患者如有多条记录，将按ICU记录ID排序保留第一条。")
    }
    ord=order(d$.patient,d$.admission,d$.stay)
    keep=sort(ord[!duplicated(d$.patient[ord])])
    rule=if(has_time) "Earliest available eligible record per patient using supplied ICU admission time" else
      "Earliest input record per patient; ICU admission-time audit unavailable"
  } else {
    warning(cohort,": 建模CSV缺少 ",id,"/",stay,
            "；无法在本代码内核验重复患者或首次ICU记录。假定上游提取已完成去重，并保留全部行。")
    d$.patient=paste0(cohort,"_row_",d$.source_row)
    d$.stay=d$.patient
    d$.admission=as.POSIXct(rep(NA_real_,nrow(d)),origin="1970-01-01",tz="UTC")
    keep=seq_len(nrow(d))
    rule="All supplied rows retained; patient/stay identifiers unavailable, so upstream deduplication cannot be audited"
  }
  audit=data.frame(Cohort=cohort,Source_row=d$.source_row,Patient_ID=d$.patient,
                   Stay_ID=d$.stay,Admission_time=as.character(d$.admission),Retained=seq_len(nrow(d)) %in% keep,
                   Rule=rule)
  list(data=d[keep,,drop=FALSE],audit=audit)
}
r6_imputer_fit <- function(x) {
  missing=vapply(x,function(z)mean(is.na(z)),numeric(1))
  keep=names(x)[missing<=MISSING_LIMIT & missing<1]
  fill=vapply(keep,function(nm) {
    z=na.omit(x[[nm]])
    if(nm %in% BINARY_FEATURES) as.numeric(names(sort(table(z),decreasing=TRUE))[1]) else median(z)
  },numeric(1))
  obj=list(keep=keep,fill=fill,missing=missing)
  a=r6_imputer_apply(obj,x)
  nonconstant=vapply(a,function(z)length(unique(z))>1L,logical(1))
  obj$keep=keep[nonconstant];obj$fill=fill[obj$keep]
  if(length(obj$keep)<2L) stop("训练部分可用变量少于2项")
  obj
}
r6_imputer_apply <- function(obj,x) {
  x=x[,obj$keep,drop=FALSE]
  for(nm in obj$keep) x[[nm]][is.na(x[[nm]])]=obj$fill[[nm]]
  if(anyNA(x)) stop("插补后仍缺失")
  x
}
r6_folds <- function(d,k,seed) {
  if(anyDuplicated(d$.patient)) stop("分折前必须先处理重复患者")
  if(min(table(d$Result))<k) stop("少数类别例数少于折数")
  set.seed(seed); caret::createFolds(d$Result,k=k,returnTrain=FALSE)
}
r6_auc <- function(y,p) {
  if(anyNA(p) || any(!is.finite(p)) || any(p<0 | p>1)) stop("预测概率无效")
  as.numeric(pROC::auc(pROC::roc(y,p,levels=c("No","Yes"),direction="<",quiet=TRUE)))
}
r6_lasso_path <- function(x,y,ratios) {
  pilot=glmnet::glmnet(as.matrix(x),as.integer(y=="Yes"),family="binomial",alpha=1,
                       standardize=TRUE,nlambda=20)
  mx=pilot$lambda[1]
  fit=glmnet::glmnet(as.matrix(x),as.integer(y=="Yes"),family="binomial",alpha=1,
                     standardize=TRUE,lambda=mx*ratios)
  list(fit=fit,lambda=mx*ratios,lambda_max=mx)
}
r6_preprocess_fit <- function(d,seed,tag) {
  if(!identical(LASSO_SELECTION_RULE,"lambda.min"))
    stop("本脚本只支持lambda.min；不要把配置标签与实际选择规则改成不一致。")
  x=d[,CANDIDATE_FEATURES,drop=FALSE]
  ratios=exp(seq(log(1),log(0.0001),length.out=60L))
  lf=r6_folds(d,LASSO_FOLDS,seed+1L)
  losses=matrix(NA_real_,length(lf),length(ratios))
  for(j in seq_along(lf)) {
    va=lf[[j]];tr=setdiff(seq_len(nrow(d)),va)
    ip=r6_imputer_fit(x[tr,,drop=FALSE])
    a=r6_imputer_apply(ip,x[tr,,drop=FALSE]);b=r6_imputer_apply(ip,x[va,,drop=FALSE])
    zz=r6_lasso_path(a,d$Result[tr],ratios)
    p=as.matrix(predict(zz$fit,newx=as.matrix(b),s=zz$lambda,type="response"))
    p=pmin(pmax(p,1e-15),1-1e-15);y=as.integer(d$Result[va]=="Yes")
    losses[j,]=colMeans(-2*(y*log(p)+(1-y)*log1p(-p)))
  }
  mu=colMeans(losses);se=apply(losses,2,sd)/sqrt(nrow(losses));best=which.min(mu)
  chosen=best
  reference_1se=which(mu<=mu[best]+se[best])[1]
  ip=r6_imputer_fit(x);a=r6_imputer_apply(ip,x)
  lp=r6_lasso_path(a,d$Result,ratios)
  co=as.matrix(coef(lp$fit,s=lp$lambda[chosen]))
  lasso=setdiff(rownames(co)[co[,1]!=0],"(Intercept)")
  set.seed(seed)
  bo=Boruta::Boruta(x=a,y=d$Result,maxRuns=BORUTA_MAX_RUNS,doTrace=0,
                    holdHistory=TRUE,num.threads=2)
  if(any(bo$finalDecision=="Tentative")) bo=Boruta::TentativeRoughFix(bo)
  boruta=names(bo$finalDecision)[bo$finalDecision=="Confirmed"]
  features=names(a)[names(a) %in% intersect(boruta,lasso)]
  if(!length(features)) stop(tag,": Boruta与LASSO交集为空，停止并核对；不会静默改为并集或固定11变量")
  obj=list(imputer=ip,features=features,boruta=boruta,lasso=lasso,
           boruta_decisions=bo$finalDecision,lambda_ratio=ratios[chosen],lambda=lp$lambda[chosen],
           lasso_selection_rule=LASSO_SELECTION_RULE,lambda_min_index=best,lambda_1se_reference_index=reference_1se,
           lasso_cv=data.frame(Ratio=ratios,Mean_deviance=mu,SE=se,Selected=seq_along(ratios)==chosen),
           seed=seed,tag=tag,training_source_rows=d$.source_row,training_patients=d$.patient)
  saveRDS(obj,file.path("Pipeline_audit",paste0(tag,"_preprocess.rds")))
  write.csv(obj$lasso_cv,file.path("Pipeline_audit",paste0(tag,"_LASSO_CV.csv")),row.names=FALSE)
  write.csv(data.frame(Stage=tag,Feature=CANDIDATE_FEATURES,
                       Missing_fraction=ip$missing[CANDIDATE_FEATURES],
                       Available=CANDIDATE_FEATURES %in% ip$keep,Boruta=CANDIDATE_FEATURES %in% boruta,
                       LASSO=CANDIDATE_FEATURES %in% lasso,Intersection=CANDIDATE_FEATURES %in% features),
            file.path("Pipeline_audit",paste0(tag,"_features.csv")),row.names=FALSE)
  obj
}
r6_preprocess_apply <- function(obj,d) {
  x=r6_imputer_apply(obj$imputer,d[,CANDIDATE_FEATURES,drop=FALSE])
  data.frame(Result=d$Result,x[,obj$features,drop=FALSE],check.names=FALSE)
}

r6_fit <- function(nm,d,g,seed) {
  set.seed(seed)
  if(nm=="LightGBM") return(lightgbm::lgb.train(r6_lgb_params(g,seed),
                                                lightgbm::lgb.Dataset(as.matrix(d[,-1,drop=FALSE]),label=as.integer(d$Result=="Yes")),nrounds=as.integer(g$nrounds)))
  if(nm=="CatBoost") return(catboost::catboost.train(catboost::catboost.load_pool(
    as.matrix(d[,-1,drop=FALSE]),label=as.integer(d$Result=="Yes")),NULL,r6_cat_params(g,seed)))
  args=list(Result~.,data=d,method=unname(r6_methods[nm]),tuneGrid=g,
            trControl=caret::trainControl(method="none",classProbs=TRUE,allowParallel=FALSE))
  if(nm %in% c("SVM","KNN","NeuralNetwork")) args$preProcess=c("center","scale")
  if(nm=="Logistic") {args$tuneGrid=NULL;args$family=binomial()}
  if(nm=="GBM") args$verbose=FALSE
  if(nm=="NeuralNetwork") {args$trace=FALSE;args$maxit=500L;args$MaxNWts=10000L}
  if(nm=="Xgboost") {args$nthread=2L;args$verbose=0L}
  do.call(caret::train,args)
}
r6_predict <- function(nm,fit,d) {
  if(nm=="LightGBM") return(as.numeric(predict(fit,as.matrix(d[,-1,drop=FALSE]))))
  if(nm=="CatBoost") return(as.numeric(catboost::catboost.predict(fit,
                                                                  catboost::catboost.load_pool(as.matrix(d[,-1,drop=FALSE])),prediction_type="Probability")))
  as.numeric(predict(fit,newdata=d,type="prob")$Yes)
}

r6_pick <- function(s) {
  required <- c("Model", "Mean_CV_AUC", "SE_CV_AUC")
  if (!is.data.frame(s) || !all(required %in% names(s)) || !nrow(s))
    stop("模型选择输入为空或缺少Model/Mean_CV_AUC/SE_CV_AUC。")
  if (anyNA(s$Model) || any(!nzchar(as.character(s$Model))) || anyDuplicated(s$Model))
    stop("每个模型必须有且仅有一组已由内层CV选定的参数。")
  if (!is.numeric(s$Mean_CV_AUC) || any(!is.finite(s$Mean_CV_AUC)) ||
      any(s$Mean_CV_AUC < 0 | s$Mean_CV_AUC > 1) ||
      !is.numeric(s$SE_CV_AUC) || any(!is.finite(s$SE_CV_AUC)) || any(s$SE_CV_AUC < 0))
    stop("模型CV指标无效；请核对训练日志，不会静默忽略失败模型。")
  s$Model <- as.character(s$Model)
  if (!setequal(s$Model, PRIMARY_MODEL_CANDIDATES))
    stop("模型选择汇总必须完整包含预先登记的九个候选模型。")
  s <- s[order(-s$Mean_CV_AUC, s$Model, method = "radix"), , drop = FALSE]
  s$Overall_AUC_rank <- seq_len(nrow(s))
  s$SHAP_explainable <- s$Model %in% SHAP_EXPLAINABLE_MODELS
  s$Eligible_for_primary <- s$Model %in% PRIMARY_MODEL_CANDIDATES &
    !s$Model %in% PRIMARY_MODEL_EXCLUSIONS
  s$Primary_rank <- NA_integer_
  eligible <- which(s$Eligible_for_primary)
  if (!length(eligible)) stop("没有满足预先登记资格的候选模型。")
  s$Primary_rank[eligible] <- seq_along(eligible)
  selected <- s$Model[eligible[1L]]
  overall_leader <- s$Model[1L]
  fallback_used <- FALSE
  s$Selected_by_rule <- s$Model == selected
  s$SHAP_used_for_selection <- FALSE
  s$Primary_role <- ifelse(s$Selected_by_rule, "Selected primary model",
                           ifelse(s$Eligible_for_primary, "Eligible candidate comparator",
                                  "Not eligible for primary selection"))
  s$Selection_rule <- MODEL_SELECTION_RULE
  s$Selection_timing <- MODEL_SELECTION_TIMING
  stopifnot(selected %in% PRIMARY_MODEL_CANDIDATES,
            identical(selected,overall_leader),
            selected %in% SHAP_EXPLAINABLE_MODELS,
            !selected %in% PRIMARY_MODEL_EXCLUSIONS,
            sum(s$Selected_by_rule) == 1L)
  excluded <- setdiff(PRIMARY_MODEL_CANDIDATES, s$Model[s$Eligible_for_primary])
  list(model = selected, leader = selected, overall_leader = overall_leader,
       selected_mean_auc = s$Mean_CV_AUC[eligible[1L]],
       excluded = if(length(excluded)) paste(excluded, collapse = ";") else "None",
       fallback_used = fallback_used,
       rule = MODEL_SELECTION_RULE, audit = s)
}
r6_search <- function(d,seed,tag) {
  folds=r6_folds(d,INNER_FOLDS,seed);sc=list();count=0L
  saveRDS(lapply(folds,function(i)d$.source_row[i]),file.path("Pipeline_audit",paste0(tag,"_inner_validation_rows.rds")))
  for(j in seq_along(folds)) {
    va=folds[[j]];tr=setdiff(seq_len(nrow(d)),va)
    stopifnot(!length(intersect(d$.patient[tr],d$.patient[va])))
    stage=paste0(tag,"_Inner",j)
    pp=r6_preprocess_fit(d[tr,,drop=FALSE],seed+100L*j,stage)
    a=r6_preprocess_apply(pp,d[tr,,drop=FALSE]);b=r6_preprocess_apply(pp,d[va,,drop=FALSE])
    for(nm in names(r6_grids)) for(q in seq_len(nrow(r6_grids[[nm]]))) {
      g=r6_grids[[nm]][q,,drop=FALSE]
      fit=r6_fit(nm,a,g,seed+1000L*j+q)
      p=r6_predict(nm,fit,b);count=count+1L
      sc[[count]]=data.frame(Stage=tag,Fold=j,Model=nm,Config=q,AUC=r6_auc(b$Result,p))
    }
    message(tag,"：内层第",j,"/",length(folds),"折完成；交集",length(pp$features),"项")
  }
  scores=do.call(rbind,sc)
  summary=do.call(rbind,lapply(split(scores,list(scores$Model,scores$Config),drop=TRUE),function(z)
    data.frame(Model=z$Model[1],Config=z$Config[1],Mean_CV_AUC=mean(z$AUC),
               SD_CV_AUC=sd(z$AUC),SE_CV_AUC=sd(z$AUC)/sqrt(nrow(z)),Resamples=nrow(z))))
  summary=summary[order(summary$Model,-summary$Mean_CV_AUC,summary$Config),]
  best=summary[!duplicated(summary$Model),];pick=r6_pick(best)
  write.csv(scores,file.path("Pipeline_audit",paste0(tag,"_all_grid_fold_AUC.csv")),row.names=FALSE)
  write.csv(summary,file.path("Pipeline_audit",paste0(tag,"_grid_summary.csv")),row.names=FALSE)
  write.csv(pick$audit,file.path("Pipeline_audit",paste0(tag,"_primary_model_selection.csv")),row.names=FALSE)
  list(best=best,selection=pick,scores=scores)
}

choose_training_threshold <- function(y, p) {
  stopifnot(length(y)==length(p), !anyNA(y), !anyNA(p), all(is.finite(p)),
            all(p >= 0 & p <= 1), length(unique(y))==2L)
  r = pROC::roc(y, p, levels=c("No","Yes"), direction="<", quiet=TRUE)
  ok = which(is.finite(r$thresholds))
  if (!length(ok)) stop("无有限阈值，模型预测可能为常数")
  j = r$sensitivities + r$specificities - 1
  best = ok[abs(j[ok] - max(j[ok])) < 1e-12]
  min(r$thresholds[best])
}

r7_glm_coef <- function(x,y,offset=rep(0,length(y))) {
  warnings=character()
  fit=tryCatch(withCallingHandlers(
    stats::glm.fit(x=x,y=y,offset=offset,family=binomial(),control=glm.control(maxit=100L)),
    warning=function(w) {warnings<<-c(warnings,conditionMessage(w));invokeRestart("muffleWarning")}),
    error=function(e) NULL)
  ok=!is.null(fit) && isTRUE(fit$converged) && fit$rank==ncol(x) && all(is.finite(fit$coefficients))
  list(coef=if(ok)fit$coefficients else rep(NA_real_,ncol(x)),
       issue=if(ok && !length(warnings)) "" else paste(c(
         if(ok) "Calibration GLM warning; finite converged estimate retained" else "Calibration GLM unavailable/nonconverged",
         unique(warnings)),collapse="; "))
}
r7_calibration <- function(y,p,eps=R1C7_LOGIT_EPS,span=R1C7_LOWESS_SPAN) {
  stopifnot(length(y)==length(p),all(y %in% 0:1),!anyNA(y),!anyNA(p),all(is.finite(p)),all(p>=0&p<=1))
  lp=qlogis(pmin(pmax(p,eps),1-eps))
  intercept=slope=NA_real_;issues=character()
  if(length(unique(y))==2L) {
    a=r7_glm_coef(matrix(1,length(y),1L),y,offset=lp);intercept=a$coef[1]
    if(nzchar(a$issue)) issues=c(issues,a$issue)
    if(diff(range(lp))>1e-10) {
      separated=max(lp[y==0])<=min(lp[y==1]) || max(lp[y==1])<=min(lp[y==0])
      if(separated) issues=c(issues,"Complete/quasi separation in calibration slope model") else {
        b=r7_glm_coef(cbind(1,lp),y);slope=b$coef[2]
        if(nzchar(b$issue)) issues=c(issues,b$issue)
      }
    } else issues=c(issues,"Constant probabilities: slope undefined")
  } else issues=c(issues,"Only one outcome class: intercept/slope undefined")
  smooth=tryCatch({
    if(length(unique(p))==1L) rep(mean(y),length(p)) else {
      z=stats::lowess(p,y,f=span,iter=0L,delta=0.01*diff(range(p)))
      stats::approx(z$x,z$y,xout=p,ties=mean,rule=2)$y
    }
  },error=function(e)rep(NA_real_,length(p)))
  smooth_bounded=pmin(pmax(smooth,0),1)
  ici=if(all(is.finite(smooth_bounded))) mean(abs(smooth_bounded-p)) else NA_real_
  if(!is.finite(ici)) issues=c(issues,"LOWESS calibration curve unavailable")
  list(values=c(CITL_intercept=unname(intercept),Calibration_slope=unname(slope),
                Brier=mean((p-y)^2),ICI=ici),smooth=smooth_bounded,
       logit_clipped=sum(p<eps|p>1-eps),smooth_clipped=sum(smooth<0|smooth>1,na.rm=TRUE),
       issue=paste(unique(issues),collapse="; "))
}
r7_ci <- function(x,B,min_fraction=.95) {
  v=x[is.finite(x)];ok=length(v)>=ceiling(min_fraction*B)
  c(low=if(ok)unname(quantile(v,.025,type=7)) else NA_real_,
    high=if(ok)unname(quantile(v,.975,type=7)) else NA_real_,valid=length(v))
}
r7_export <- function(predictions,selected_model,fixed_thresholds,output_root,smoke_test=FALSE,
                      B=R1C7_BOOTSTRAPS,seed=R1C7_BOOT_SEED) {
  if(smoke_test) B=min(as.integer(B),100L)
  if(B<20L) stop("Bootstrap次数过少；正式运行默认1000次")
  if(!smoke_test && B<1000L) warning("正式CI建议至少保留本版默认的1000次；当前设置低于默认")
  models=names(predictions[[1]])[-1]
  stopifnot(selected_model %in% models,all(models %in% names(fixed_thresholds)),
            all(vapply(predictions,function(z)identical(names(z),c("Result",models)),logical(1))))
  out=file.path(output_root,"R1C7_calibration_DCA")
  if(dir.exists(out) && length(list.files(out))) {
    out=paste0(out,"_",format(Sys.time(),"%Y%m%d_%H%M%S"))
    if(dir.exists(out)) stop("第7条输出目录已存在，请更改目录或稍后再试")
  }
  dir.create(out,recursive=TRUE,showWarnings=FALSE)
  had_seed=exists(".Random.seed",envir=.GlobalEnv,inherits=FALSE)
  if(had_seed) old_seed=get(".Random.seed",envir=.GlobalEnv)
  on.exit(if(had_seed) assign(".Random.seed",old_seed,envir=.GlobalEnv) else
    if(exists(".Random.seed",envir=.GlobalEnv,inherits=FALSE)) rm(".Random.seed",envir=.GlobalEnv),add=TRUE)
  scope=if(smoke_test) "SMOKE TEST ONLY; not manuscript results" else "Formal evaluation of saved final-model predictions"
  pt=sort(unique(c(round(R1C7_DCA_THRESHOLDS,12),round(R1C7_DCA_EXAMPLES,12),unname(fixed_thresholds))))
  pt=pt[is.finite(pt)&pt>0&pt<1];w=pt/(1-pt)
  metric_rows=dca_rows=diagnostics=list();bootstrap_values=list()
  metric_count=dca_count=diagnostic_count=0L
  ideals=c(CITL_intercept=0,Calibration_slope=1,Brier=0,ICI=0)
  saveRDS(list(B=B,seed=seed,logit_epsilon=R1C7_LOGIT_EPS,lowess_span=R1C7_LOWESS_SPAN,
               lowess_iter=0L,lowess_delta="0.01 * predicted-probability range",smooth_bounded_to=c(0,1),
               thresholds=pt,examples=R1C7_DCA_EXAMPLES,action=R1C7_ACTION,action_status=R1C7_ACTION_STATUS,
               model=selected_model,smoke_test=smoke_test,
               CI="Percentile 2.5/97.5%; unstratified row bootstrap; fixed predictions; pointwise; at least 95% valid replicates"),
          file.path(out,"R1C7_configuration.rds"))
  for(di in seq_along(predictions)) {
    ds=names(predictions)[di];z=predictions[[di]];n=nrow(z)
    if(!all(as.character(z$Result) %in% c("No","Yes"))) stop(ds,": Result需要No/Yes")
    y=as.integer(z$Result=="Yes");rho=mean(y)
    if(length(unique(y))<2L) stop(ds,": 整个队列只有一种结局，无法完整评价")
    role=if(ds=="Train") "Apparent training performance; no optimism correction" else "Evaluation cohort; conditional on fitted models"
    set.seed(seed+di)
    idx=matrix(sample.int(n,n*B,replace=TRUE),nrow=n,ncol=B)
    counts=vapply(seq_len(B),function(b)tabulate(idx[,b],nbins=n),integer(n))
    storage.mode(counts)="double"
    prevalence_boot=as.vector(crossprod(y,counts))/n
    nb_all=rho-(1-rho)*w
    all_boot=outer(rep(1,length(pt)),prevalence_boot)-outer(w,1-prevalence_boot)
    for(m in models) {
      p=z[[m]];point=r7_calibration(y,p)
      boot=matrix(NA_real_,nrow=B,ncol=4L,dimnames=list(NULL,names(point$values)))
      issue_count=0L
      for(b in seq_len(B)) {
        q=r7_calibration(y[idx[,b]],p[idx[,b]])
        boot[b,]=q$values
        if(nzchar(q$issue)) issue_count=issue_count+1L
      }
      bootstrap_values[[paste(ds,m,sep="::")]]=boot
      for(metric in names(point$values)) {
        ci=r7_ci(boot[,metric],B);metric_count=metric_count+1L
        point_valid=is.finite(point$values[[metric]])
        if(!point_valid) ci[c("low","high")]=NA_real_
        metric_rows[[metric_count]]=data.frame(DataSet=ds,Model=m,N=n,Events=sum(y),Prevalence=rho,
                                               Metric=metric,Estimate=point$values[[metric]],CI_low=ci["low"],CI_high=ci["high"],
                                               Ideal=ideals[[metric]],Bootstrap_requested=B,Bootstrap_valid=ci["valid"],
                                               Status=if(point_valid && all(is.finite(ci[c("low","high")]))) "OK" else "CHECK: estimate or CI unavailable",
                                               Evaluation_role=role,Run_scope=scope,row.names=NULL)
      }
      diagnostic_count=diagnostic_count+1L
      diagnostics[[diagnostic_count]]=data.frame(DataSet=ds,Model=m,Probabilities_clipped_for_logit=point$logit_clipped,
                                                 LOWESS_values_clipped_to_unit_interval=point$smooth_clipped,Point_estimate_issues=point$issue,
                                                 Bootstrap_replicates_with_calibration_issues=issue_count,Run_scope=scope)
      if(m==selected_model) write.csv(data.frame(Predicted_probability=p,LOWESS_calibrated_probability=point$smooth),
                                      file.path(out,paste0(ds,"_Selected_model_LOWESS_curve.csv")),row.names=FALSE)
      high=outer(p,pt,`>`)
      tp=high*y;fp=high*(1-y)
      tp_boot=crossprod(tp,counts)/n;fp_boot=crossprod(fp,counts)/n
      nb=colMeans(tp)-colMeans(fp)*w
      nb_boot=tp_boot-fp_boot*w
      delta_all=nb_boot-all_boot
      for(j in seq_along(pt)) {
        ci=r7_ci(nb_boot[j,],B);ci_all=r7_ci(delta_all[j,],B)
        snb_ci=r7_ci(nb_boot[j,]/prevalence_boot,B)
        dca_count=dca_count+1L
        dca_rows[[dca_count]]=data.frame(DataSet=ds,Model=m,Threshold=pt[j],
                                         Net_benefit=nb[j],NB_CI_low=ci["low"],NB_CI_high=ci["high"],
                                         Standardized_NB=nb[j]/rho,sNB_CI_low=snb_ci["low"],sNB_CI_high=snb_ci["high"],
                                         Assess_all_NB=nb_all[j],Assess_none_NB=0,
                                         Difference_vs_all=nb[j]-nb_all[j],Difference_vs_all_CI_low=ci_all["low"],Difference_vs_all_CI_high=ci_all["high"],
                                         Difference_vs_none=nb[j],Difference_vs_none_CI_low=ci["low"],Difference_vs_none_CI_high=ci["high"],
                                         TP=sum(tp[,j]),FP=sum(fp[,j]),Additional_assessments_per_100=100*mean(high[,j]),
                                         False_positive_weight=w[j],False_positives_per_true_positive=(1-pt[j])/pt[j],
                                         Is_illustrative_threshold=any(abs(pt[j]-R1C7_DCA_EXAMPLES)<1e-12),
                                         Is_models_fixed_classification_threshold=abs(pt[j]-fixed_thresholds[[m]])<1e-12,
                                         Bootstrap_valid=ci["valid"],DCA_positive_rule="Probability > threshold (rmda convention)",
                                         Evaluation_role=role,Run_scope=scope,row.names=NULL)
      }
      message("第7条：",ds," / ",m," 的校准与净获益区间完成（",B,"次bootstrap）")
    }
  }
  cal=do.call(rbind,metric_rows);dca=do.call(rbind,dca_rows);diag=do.call(rbind,diagnostics)
  write.csv(cal,file.path(out,"R1C7_Calibration_all_models_95CI.csv"),row.names=FALSE)
  write.csv(diag,file.path(out,"R1C7_Calibration_diagnostics.csv"),row.names=FALSE)
  saveRDS(bootstrap_values,file.path(out,"R1C7_Calibration_bootstrap_values.rds"))
  selected=subset(cal,Model==selected_model & DataSet!="Train")
  write.csv(selected,file.path(out,"R1C7_Selected_model_calibration_95CI.csv"),row.names=FALSE)
  formatted=do.call(rbind,lapply(split(selected,selected$DataSet),function(z) {
    r=data.frame(Cohort=z$DataSet[1],Model=selected_model)
    for(mm in names(ideals)) {
      zz=z[z$Metric==mm,]
      r[[mm]]=if(all(is.finite(unlist(zz[c("Estimate","CI_low","CI_high")]))))
        sprintf("%.3f (%.3f to %.3f)",zz$Estimate,zz$CI_low,zz$CI_high) else "Not estimable; see diagnostics"
    }
    r
  }))
  write.csv(formatted,file.path(out,"R1C7_Selected_model_table_for_manuscript.csv"),row.names=FALSE)
  write.csv(dca,file.path(out,"R1C7_DCA_all_models_pointwise_95CI.csv"),row.names=FALSE)
  write.csv(subset(dca,Model==selected_model & DataSet!="Train" & Is_illustrative_threshold),
            file.path(out,"R1C7_Selected_model_DCA_illustrative_thresholds.csv"),row.names=FALSE)
  write.csv(subset(dca,Is_models_fixed_classification_threshold),
            file.path(out,"R1C7_DCA_at_fixed_classification_thresholds.csv"),row.names=FALSE)
  action=data.frame(Threshold=R1C7_DCA_EXAMPLES,
                    Trigger=paste0("Predicted subsequent-sepsis risk > ",100*R1C7_DCA_EXAMPLES,"% at 24h landmark"),
                    Action=R1C7_ACTION,FP_weight=R1C7_DCA_EXAMPLES/(1-R1C7_DCA_EXAMPLES),
                    FP_per_TP=(1-R1C7_DCA_EXAMPLES)/R1C7_DCA_EXAMPLES,Status=R1C7_ACTION_STATUS)
  write.csv(action,file.path(out,"R1C7_DCA_action_and_threshold_interpretation.csv"),row.names=FALSE)
  writeLines(c(
    paste("Run scope:",scope),paste("Bootstrap replicates:",B),
    "Intercept = calibration-in-the-large from logit P(Y=1)=a+offset(logit(original risk)); slope fixed at 1.",
    "Slope = b from logit P(Y=1)=a+b*logit(original risk), with intercept freely estimated.",
    "Probabilities clipped to [1e-6,1-1e-6] only for logit-based metrics; original risks retained for Brier, ICI and DCA.",
    "ICI = mean absolute difference between original probabilities and bounded LOWESS calibrated probabilities (span 2/3, iter=0).",
    "ICI smoothing is refitted within each bootstrap resample. This is a calibration diagnostic, not recalibration of model predictions.",
    "CIs are percentile intervals conditional on fitted models; no full development-pipeline refitting or optimism correction is performed here.",
    "Bootstrap samples are unstratified to allow event prevalence to vary; models within a cohort share bootstrap indices.",
    "CI unavailable if fewer than 95% of replicates are finite. Consult diagnostics rather than replacing failures with zero.",
    "Numerical GLM warnings are logged; finite, full-rank converged fits are retained unless complete/quasi separation is detected.",
    "Training entries are apparent performance; do not describe them as independent validation.",
    "Record bootstrap assumes independent units. MIMIC has one retained record per subject; eICU person-level repeats remain unverified.",
    "Brier reflects overall probability accuracy, not calibration alone; its magnitude also depends on event prevalence.",
    paste("Proposed clinical action:",R1C7_ACTION),paste("Clinical status:",R1C7_ACTION_STATUS),
    "At any threshold the action is the same; the threshold changes the assumed harm/benefit trade-off and the patients alerted.",
    "Assess all = additional model-independent review for all; assess none = no model-triggered additional review. Routine ICU care is never withheld.",
    "No prophylactic antibiotics or other treatment is automatically initiated solely from this predicted future risk.",
    "0.10/0.20/0.30 are illustrative scenarios, not validated clinical cutoffs or a data-selected optimal range.",
    "The fixed Youden classification threshold is a statistical operating point, not automatically a clinical intervention threshold.",
    "DCA uses existing unrecalibrated probabilities. NB and pairwise NB differences have pointwise, not simultaneous, CIs.",
    "DCA follows rmda: risk > threshold. Table 2 retains risk >= fixed threshold; the boundary convention differs only at exact ties.",
    "Do not choose thresholds by maximizing external-cohort net benefit. No threshold recommendation is generated.",
    "Net benefit is a modeled utility under assumed preferences; it does not demonstrate prevented sepsis, improved survival, or a causal treatment effect.",
    "Existing three-bin calibration and DCA plots are retained unchanged; the new CIs are in separate tables, not added to the old plots."
  ),file.path(out,"R1C7_Methods_and_interpretation.txt"))
  if(any(cal$Status!="OK")) warning("部分校准指标/CI不可估计，详见R1C7_Calibration_diagnostics.csv")
  message("第7条补充完成：",normalizePath(out,winslash="/"))
  invisible(out)

  shap_file <- file.path(result_dir, paste0("SHAP_", model, "_values.rds"))
  if (!file.exists(shap_file)) stop("找不到已保存的SHAP结果：", shap_file,
                                    "\n请确认选择了本次分析结果文件夹，且已运行SHAP部分。")
  saved <- readRDS(shap_file)
  sv <- if (inherits(saved, "shapviz")) saved else saved[["Yes"]]
  if (!inherits(sv, "shapviz")) stop("SHAP文件不是shapviz对象，也没有Yes类别对象。")
  if (!is.matrix(sv$S) || any(!is.finite(sv$S)) || !nrow(sv$S) || !ncol(sv$S))
    stop("SHAP矩阵为空或包含无效值。")
  if (!identical(colnames(sv$S), names(sv$X)) || nrow(sv$S) != nrow(sv$X))
    stop("SHAP矩阵与特征数据不一致。")
  if (anyDuplicated(names(sv$X))) stop("特征名称重复。")
  n <- nrow(sv$S)
  if (is.null(case_rows)) {
    selected_case_file <- file.path(result_dir,"SHAP_selected_case.csv")
    if (!file.exists(selected_case_file))
      stop("找不到SHAP_selected_case.csv，无法自动读取目标概率病例。")
    selected_case <- read.csv(selected_case_file,check.names=FALSE)
    if (!all(c("Model","Training_row","Target_probability","Predicted_probability") %in% names(selected_case)) ||
        nrow(selected_case)!=1L || selected_case$Model[1]!=model)
      stop("SHAP_selected_case.csv与当前主模型或目标概率配置不一致。")
    case_rows <- selected_case$Training_row[1]
  }
  if (!is.numeric(case_rows) || anyNA(case_rows) ||
      any(!is.finite(case_rows) | case_rows != floor(case_rows) | case_rows < 1 | case_rows > n))
    stop("CASE_ROWS必须是1至", n, "之间的整数行号；只导出选样表可用integer(0)。")
  case_rows <- unique(as.integer(case_rows))
  valid_display <- function(x) is.numeric(x) && length(x) == 1L && !is.na(x) &&
    x >= 1 && (is.infinite(x) || x == floor(x))
  if (!valid_display(force_max_display) || !valid_display(waterfall_max_display))
    stop("max_display必须是正整数或Inf。")
  sizes <- c(force_width, force_height, waterfall_width, waterfall_height)
  if (!is.numeric(sizes) || length(sizes) != 4L || any(!is.finite(sizes) | sizes <= 0))
    stop("图片宽度和高度必须为正数，单位是英寸。")
  importance <- shapviz::sv_importance(sv, kind = "no")
  features <- names(importance)
  if (!is.null(case_features)) {
    if (!is.character(case_features) || !length(case_features) || anyNA(case_features))
      stop("CASE_FEATURES应为NULL，或一个非空的特征名称向量。")
    case_features <- unique(case_features)
    if (!all(case_features %in% features))
      stop("这些特征不在当前模型中：", paste(setdiff(case_features, features), collapse = ", "))
  }
  ranking <- data.frame(Rank = seq_along(features), Feature = features,
                        Mean_abs_SHAP = as.numeric(importance))
  index <- data.frame(SHAP_row = seq_len(n), SHAP_output = rowSums(sv$S) + sv$baseline)
  index <- cbind(index, sv$X)
  mapping_note <- "SHAP_row仅指该SHAP对象的行号，不是患者ID或原始CSV行号。"
  results_file <- file.path(result_dir, "R1C4_predictions_and_results.rds")
  dev_file <- file.path(result_dir, "dev.csv")
  if (file.exists(results_file) && file.exists(dev_file)) {
    rr <- readRDS(results_file)
    train <- rr$predictions$Train
    dd <- read.csv(dev_file, check.names = FALSE)
    aligned <- is.data.frame(train) && nrow(train) == nrow(dd) && nrow(dd) >= n &&
      all(names(sv$X) %in% names(dd)) && model %in% names(train)
    if (aligned) aligned <- all(vapply(names(sv$X), function(nm) {
      a <- sv$X[[nm]]; b <- dd[[nm]][seq_len(n)]
      if (is.numeric(a)) isTRUE(all.equal(a, as.numeric(b), tolerance = 1e-7,
                                          check.attributes = FALSE)) else
                                            identical(as.character(a), as.character(b))
    }, logical(1)))
    if (!aligned) stop("SHAP对象与dev.csv/训练预测的行顺序或内容不一致，请勿混用不同运行结果。")
    scale <- "probability"
    expected <- index$SHAP_output
    prob <- train[[model]][seq_len(n)]
    if (any(!is.finite(prob)) || max(abs(expected - prob)) > 1e-6)
      stop("SHAP加和与已保存预测不一致，请检查模型、解释尺度及结果来源。")
    index <- cbind(data.frame(SHAP_row = seq_len(n), Training_row = seq_len(n),
                              Observed_Result = train$Result[seq_len(n)],
                              Predicted_probability = prob, SHAP_scale = scale),
                   index[, setdiff(names(index), "SHAP_row"), drop = FALSE])
    mapping_note <- paste("已核验特征与训练预测；SHAP_row对应dev.csv的数据行（不计表头）。",
                          "SHAP坐标尺度：", scale, "。")
    manifest_file <- file.path(result_dir, "MIMICIV_split_manifest.csv")
    if (file.exists(manifest_file)) {
      mm <- read.csv(manifest_file, colClasses = "character", check.names = FALSE)
      need <- c("Source_row", "Patient_ID", "Stay_ID", "Split")
      if (all(need %in% names(mm)) &&
          identical(as.integer(which(mm$Split == "Train")), as.integer(rr$split_indices)) &&
          sum(mm$Split == "Train") == nrow(dd)) {
        id_columns <- mm[mm$Split == "Train", c("Patient_ID", "Stay_ID", "Source_row"), drop = FALSE]
        index <- cbind(index[, 1:2], id_columns[seq_len(n), , drop = FALSE],
                       index[, -(1:2), drop = FALSE])
      }
    }
  }
  case_sv <- sv
  grouped <- character(0)
  if (!is.null(case_features)) {
    grouped <- setdiff(features, case_features)
    S_case <- sv$S[, case_features, drop = FALSE]
    X_case <- sv$X[, case_features, drop = FALSE]
    if (length(grouped)) {
      group_name <- "Other features"
      while (group_name %in% features) group_name <- paste0(group_name, "_")
      S_case <- cbind(S_case, rowSums(sv$S[, grouped, drop = FALSE]))
      colnames(S_case)[ncol(S_case)] <- group_name
      X_case[[group_name]] <- "combined"
    }
    case_sv <- shapviz::shapviz(S_case, X = X_case, baseline = sv$baseline)
    stopifnot(max(abs(rowSums(case_sv$S) - rowSums(sv$S))) < 1e-10)
    force_max_display <- Inf
    waterfall_max_display <- Inf
  }
  dir.create(output_parent, recursive = TRUE, showWarnings = FALSE)
  out <- file.path(output_parent, paste0("SHAP_custom_", model, "_", format(Sys.time(), "%Y%m%d_%H%M%S")))
  initial_out <- out; suffix <- 1L
  while (file.exists(out)) { out <- paste0(initial_out, "_", suffix); suffix <- suffix + 1L }
  if (!dir.create(out, recursive = TRUE)) stop("无法创建输出文件夹：", out)
  write.csv(ranking, file.path(out, "SHAP_feature_ranking.csv"), row.names = FALSE)
  write.csv(index, file.path(out, "SHAP_sample_index.csv"), row.names = FALSE)
  plot_pdf <- function(p, file, width, height) {
    grDevices::pdf(file, width = width, height = height)
    on.exit(grDevices::dev.off(), add = TRUE)
    print(p)
  }
  file_safe <- function(x) gsub("[^A-Za-z0-9_.-]", "_", x)
  if (isTRUE(draw_dependence)) for (j in seq_along(features)) {
    v <- features[j]
    p <- shapviz::sv_dependence(sv, v = v, color = "#3b528b", color_var = v) +
      ggplot2::theme_bw() + ggplot2::ggtitle(model) +
      ggplot2::theme(plot.title = ggplot2::element_text(hjust = 0.5, face = "bold", color = "black"))
    plot_pdf(p, file.path(out, sprintf("SHAP_%s_dependence_%02d_%s.pdf", model, j, file_safe(v))), 5, 5)
  }
  case_details <- list()
  for (r in case_rows) {
    stem <- sprintf("SHAP_%s_case_%04d", model, r)
    p <- shapviz::sv_force(case_sv, row_id = r, max_display = force_max_display, size = 9) +
      ggplot2::ggtitle(model) +
      ggplot2::theme(plot.margin = ggplot2::margin(5.5, 12, 5.5, 12),
                     plot.title = ggplot2::element_text(hjust = 0.5, face = "bold", color = "black"))
    plot_pdf(p, file.path(out, paste0(stem, "_force.pdf")), force_width, force_height)
    p <- shapviz::sv_waterfall(case_sv, row_id = r, max_display = waterfall_max_display,
                               fill_colors = c("#f7d13d", "#a52c60")) +
      ggplot2::theme_bw() + ggplot2::ggtitle(model) +
      ggplot2::theme(plot.title = ggplot2::element_text(hjust = 0.5, face = "bold", color = "black"))
    plot_pdf(p, file.path(out, paste0(stem, "_waterfall.pdf")), waterfall_width, waterfall_height)
    ord <- order(abs(sv$S[r, ]), decreasing = TRUE)
    case_details[[as.character(r)]] <- data.frame(SHAP_row = r,
                                                  Feature = colnames(sv$S)[ord], Value = vapply(sv$X[ord], function(x) as.character(x[r]), character(1)),
                                                  SHAP_value = as.numeric(sv$S[r, ord]),
                                                  Baseline = sv$baseline, SHAP_output = sum(sv$S[r, ]) + sv$baseline)
  }
  if (length(case_details)) write.csv(do.call(rbind, case_details),
                                      file.path(out, "SHAP_selected_case_values.csv"), row.names = FALSE)
  smoke_file <- file.path(result_dir, "Run_configuration.rds")
  smoke <- file.exists(smoke_file) && isTRUE(readRDS(smoke_file)$smoke_test)
  writeLines(c(if (smoke) "SMOKE TEST ONLY：此文件夹的来源是调试分析，不能作为正式结果。" else "来源：用户指定的已保存SHAP结果。",
               paste("Model:", model), paste("Source SHAP:", shap_file), paste("SHAP samples:", n),
               paste("Explained features:", length(features)), mapping_note,
               paste("Target predicted probability for the waterfall case:",SHAP_TARGET_PROBABILITY),
               "The selected case is the real training observation whose predicted probability is closest to the target; f(x) is not overwritten.",
               "依赖图是SHAP dependence plot，不是偏依赖图/PDP，也不是因果效应图。",
               "仅遍历该模型SHAP对象内的全部入选特征，不包含被筛除的候选特征。",
               "默认格式沿用最新代码的主模型SHAP图；所有新增图单独保存，不覆盖原图。",
               "CASE_ROWS一次循环一人；多个行号不会求平均。",
               paste("Explicit case features:", if (is.null(case_features)) "NULL (all available)" else paste(case_features, collapse = ", ")),
               paste("Grouped as Other features:", paste(grouped, collapse = ", ")),
               "指定部分特征时，其余SHAP贡献合并为Other features；不丢弃贡献，不改变预测。",
               "展示案例应明确为解释性示例，不应因选择了某病例而宣称模型总体准确。"),
             file.path(out, "SHAP_plot_notes.txt"), useBytes = TRUE)
  saveRDS(list(model = model, case_rows = case_rows, case_features = case_features,
               force_max_display = force_max_display, waterfall_max_display = waterfall_max_display,
               target_probability = SHAP_TARGET_PROBABILITY,
               source_md5 = tools::md5sum(shap_file), grouped_features = grouped,
               shapviz_version = as.character(utils::packageVersion("shapviz"))),
          file.path(out, "SHAP_plot_configuration.rds"))
  message("SHAP补画完成：", normalizePath(out, winslash = "/"))
  invisible(list(output_dir = out, model = model, ranking = ranking,
                 sample_index = index, case_sv = case_sv))
}

redraw_lambda_min_selection <- function(result_dir, raw_file, output_dir,
                                        draw_venn = TRUE, confirmed_only = TRUE,
                                        x_axis = "reference_lambda", text_scale = 1,
                                        boruta_width = 5.306, boruta_height = 3.931,
                                        lasso1_width = 5.5, lasso1_height = 5.5,
                                        lasso2_width = 7.084, lasso2_height = 5.903) {
  rng_before <- RNGkind()
  had_seed <- exists(".Random.seed", envir = .GlobalEnv, inherits = FALSE)
  if (had_seed) seed_before <- get(".Random.seed", envir = .GlobalEnv)
  on.exit({
    do.call(RNGkind, as.list(rng_before))
    if (had_seed) assign(".Random.seed", seed_before, envir = .GlobalEnv)
    else if (exists(".Random.seed", envir = .GlobalEnv, inherits = FALSE))
      rm(list = ".Random.seed", envir = .GlobalEnv)
  }, add = TRUE)
  RNGkind("Mersenne-Twister", "Inversion", "Rejection")
  fail <- function(...) stop(..., call. = FALSE)
  eq_num <- function(a, b, tol = 1e-10) {
    isTRUE(all.equal(as.numeric(a), as.numeric(b), tolerance = tol,
                     check.attributes = FALSE))
  }
  if (!x_axis %in% c("reference_lambda", "log_ratio"))
    fail("LASSO_X_AXIS只能是reference_lambda或log_ratio。")
  if (!is.numeric(text_scale) || length(text_scale) != 1L ||
      !is.finite(text_scale) || text_scale <= 0) fail("TEXT_SCALE必须大于0。")
  required <- c("glmnet", "Boruta", "fru", "ggplot2")
  if (draw_venn) required <- c(required, "VennDiagram")
  missing <- required[!vapply(required, requireNamespace, logical(1), quietly = TRUE)]
  if (length(missing)) fail("缺少R包：", paste(missing, collapse = ", "),
                            "。请使用原分析的软件环境；脚本不会自动安装或升级。")
  paths <- c(preprocess = file.path(result_dir, "Final_preprocess.rds"),
             config = file.path(result_dir, "Run_configuration.rds"),
             candidates = file.path(result_dir, "Candidate_predictors.csv"),
             split = file.path(result_dir, "MIMICIV_split_manifest.csv"),
             features = file.path(result_dir, "Pipeline_audit", "FinalRefit_features.csv"),
             dev = file.path(result_dir, "dev.csv"),
             session = file.path(result_dir, "sessionInfo.txt"),
             raw = raw_file)
  if (any(!file.exists(paths))) fail("缺少文件：\n", paste(paths[!file.exists(paths)], collapse = "\n"))
  source_hashes <- setNames(tools::md5sum(unname(paths)), names(paths))
  pp <- readRDS(paths[["preprocess"]])
  cfg <- readRDS(paths[["config"]])
  if (isTRUE(cfg$smoke_test)) fail("该目录是SMOKE_TEST调试结果，不能生成论文最终筛选图。")
  if (identical(cfg$analysis_mode, "FIXED11")) fail("该目录是固定11项分析；请改为lambda.min完整折内筛选结果目录。")
  if (!identical(cfg$lasso_selection_rule, "lambda.min") ||
      !identical(pp$lasso_selection_rule, "lambda.min"))
    fail("结果不是本次lambda.min主代码生成的完整折内筛选结果。请先从头重跑主代码，不能修改旧RDS标签绕过检查。")
  fields <- c("imputer", "features", "boruta", "lasso", "boruta_decisions",
              "lambda_ratio", "lambda", "lasso_cv", "seed", "training_source_rows", "training_patients")
  if (!all(fields %in% names(pp))) fail("保存对象缺少完整筛选字段，不适用于本脚本。")
  if (!length(pp$features) || anyNA(pp$features) || anyDuplicated(pp$features) ||
      !all(pp$features %in% pp$imputer$keep))
    fail("保存的最终变量名单为空、重复或不属于可用候选变量。")
  if (!setequal(intersect(pp$boruta, pp$lasso), pp$features))
    fail("保存的Boruta/LASSO交集与最终变量名单不一致。")
  old_session <- paste(readLines(paths[["session"]], warn = FALSE), collapse = " ")
  versions <- data.frame(Package = required,
                         Current_version = vapply(required, function(p) as.character(utils::packageVersion(p)), character(1)))
  versions$Original_version <- vapply(required, function(p) {
    token <- regmatches(old_session, regexec(paste0("\\b", p, "_([^[:space:]]+)"), old_session))[[1]]
    if (length(token) >= 2L) token[2] else NA_character_
  }, character(1))
  normalize_version <- function(z) gsub("-", ".", z, fixed = TRUE)
  core <- versions$Package %in% c("glmnet", "Boruta", "fru")
  mismatch <- core & (is.na(versions$Original_version) |
                        normalize_version(versions$Original_version) != normalize_version(versions$Current_version))
  if (any(mismatch)) fail("用于重现筛选的包版本与原分析不一致：\n",
                          paste(capture.output(print(versions[mismatch, ], row.names = FALSE)), collapse = "\n"),
                          "\n请恢复原分析版本，不要用不同版本生成另一套筛选结果。")
  candidate_info <- read.csv(paths[["candidates"]], check.names = FALSE, stringsAsFactors = FALSE)
  if (!all(c("Feature", "Binary") %in% names(candidate_info))) fail("候选变量文件格式不正确。")
  candidate_features <- as.character(candidate_info$Feature)
  binary_features <- candidate_features[as.logical(candidate_info$Binary)]
  if (!identical(candidate_features, cfg$candidate_features)) fail("候选名单与原运行配置不一致。")
  if (anyNA(candidate_info$Binary) || anyDuplicated(candidate_features)) fail("候选变量的类型或名称无效。")
  saved_hash_names <- basename(gsub("\\\\", "/", names(cfg$source_md5)))
  source_idx <- which(saved_hash_names == "mimic4.csv")
  if (length(source_idx) != 1L ||
      !identical(unname(as.character(cfg$source_md5[source_idx])), unname(as.character(source_hashes[["raw"]]))))
    fail("mimic4.csv与完成本次lambda.min分析时的原始文件不一致。补图前应恢复同一输入，不能混用修改后的数据。")
  raw <- read.csv(raw_file, check.names = FALSE, na.strings = c("", "NA", "NaN", "NULL"),
                  strip.white = TRUE, fileEncoding = "GB18030")
  if ("HGB" %in% names(raw) && !"HB" %in% names(raw)) names(raw)[names(raw) == "HGB"] <- "HB"
  required_cols <- c("Result", candidate_features)
  if (!all(required_cols %in% names(raw))) fail("原始数据缺列：", paste(setdiff(required_cols, names(raw)), collapse = ", "))
  rows <- pp$training_source_rows
  if (anyNA(rows) || anyDuplicated(rows) || any(rows < 1 | rows > nrow(raw)) ||
      any(rows != as.integer(rows))) fail("保存的训练行号无效。")
  train <- raw[as.integer(rows), , drop = FALSE]
  has_raw_ids <- all(c("subject_id", "stay_id") %in% names(raw))
  if (has_raw_ids) {
    expected_patient <- as.character(train$subject_id)
    expected_stay <- as.character(train$stay_id)
  } else {
    expected_patient <- paste0("MIMIC-IV_row_", rows)
    expected_stay <- expected_patient
  }
  if (!identical(expected_patient, as.character(pp$training_patients)) ||
      anyDuplicated(expected_patient)) fail("训练记录或顺序与保存的最终筛选对象不一致。")
  split <- read.csv(paths[["split"]], colClasses = "character", check.names = FALSE)
  ii <- match(as.character(rows), split$Source_row)
  if (anyNA(ii) || !all(split$Split[ii] == "Train") ||
      !identical(expected_stay, split$Stay_ID[ii]) ||
      !identical(expected_patient, split$Patient_ID[ii]) ||
      !setequal(as.character(rows), split$Source_row[split$Split == "Train"]))
    fail("训练记录与MIMIC-IV拆分清单不一致。程序不会重新随机拆分。")
  for (nm in c("Result", candidate_features)) {
    old <- train[[nm]]
    z <- suppressWarnings(as.numeric(as.character(old)))
    if (any(is.na(z) & !is.na(old)) || any(is.infinite(z))) fail("训练数据存在非数值内容或无穷值：", nm)
    train[[nm]] <- z
  }
  if (anyNA(train$Result) || !all(train$Result %in% 0:1)) fail("训练结局不是完整的0/1。")
  for (nm in binary_features)
    if (!all(na.omit(train[[nm]]) %in% 0:1)) fail("二分类变量编码不是0/1：", nm)
  y <- factor(train$Result, levels = 0:1, labels = c("No", "Yes"))
  x_raw <- train[, candidate_features, drop = FALSE]
  missing_rate <- vapply(x_raw, function(z) mean(is.na(z)), numeric(1))
  keep <- names(x_raw)[missing_rate <= cfg$missing_limit & missing_rate < 1]
  fills <- vapply(keep, function(nm) {
    z <- na.omit(x_raw[[nm]])
    if (nm %in% binary_features) as.numeric(names(sort(table(z), decreasing = TRUE))[1]) else median(z)
  }, numeric(1))
  tmp <- x_raw[, keep, drop = FALSE]
  for (nm in keep) tmp[[nm]][is.na(tmp[[nm]])] <- fills[[nm]]
  varying <- vapply(tmp, function(z) length(unique(z)) > 1L, logical(1))
  keep <- keep[varying]
  fills <- fills[keep]
  if (!identical(keep, pp$imputer$keep) || !eq_num(fills, pp$imputer$fill) ||
      !eq_num(missing_rate, pp$imputer$missing)) fail("最终训练集的缺失处理无法重现保存对象。")
  x <- x_raw[, pp$imputer$keep, drop = FALSE]
  for (nm in names(x)) x[[nm]][is.na(x[[nm]])] <- pp$imputer$fill[[nm]]
  if (anyNA(x)) fail("使用原插补值后仍有缺失。")
  old_dev <- read.csv(paths[["dev"]], check.names = FALSE, stringsAsFactors = FALSE)
  if (nrow(old_dev) != nrow(x) || !identical(as.character(old_dev$Result), as.character(y)) ||
      !all(pp$features %in% names(old_dev)) ||
      !eq_num(as.matrix(old_dev[, pp$features, drop = FALSE]), as.matrix(x[, pp$features, drop = FALSE])))
    fail("最终输入或结局与已用于建模的dev.csv不一致。")
  message("原始文件、", nrow(x), "条训练记录和插补结果核对通过；不使用内部测试/外部队列拟合。")
  cv <- pp$lasso_cv

      anyNA(cv) || any(!is.finite(as.matrix(cv[, c("Ratio", "Mean_deviance", "SE")]))))
    fail("保存的LASSO辅助CV曲线无效。")
  if (!eq_num(cv$Ratio, exp(seq(log(1), log(0.0001), length.out = 60L))) ||
      any(cv$SE < 0) || sum(as.logical(cv$Selected)) != 1L) fail("LASSO搜索网格或选择标记与本版流程不符。")
  index_min <- which.min(cv$Mean_deviance)
  index_1se <- which(cv$Mean_deviance <= cv$Mean_deviance[index_min] + cv$SE[index_min])[1]
  index_selected <- index_min
  if (!identical(index_selected, which(as.logical(cv$Selected))) ||
      !eq_num(cv$Ratio[index_selected], pp$lambda_ratio))
    fail("保存的LASSO选择不符合lambda.min最小平均二项偏差规则。")
  pilot <- glmnet::glmnet(as.matrix(x), as.integer(y == "Yes"), family = "binomial",
                          alpha = 1, standardize = TRUE, nlambda = 20)
  lambda_max <- pilot$lambda[1]
  lambda_grid <- lambda_max * cv$Ratio
  lasso_fit <- glmnet::glmnet(as.matrix(x), as.integer(y == "Yes"), family = "binomial",
                              alpha = 1, standardize = TRUE, lambda = lambda_grid)
  coef_path <- as.matrix(stats::coef(lasso_fit, s = lambda_grid))
  coef_selected <- coef_path[, index_selected]
  lasso_names <- setdiff(rownames(coef_path)[coef_selected != 0], "(Intercept)")
  if (!eq_num(lambda_grid[index_selected], pp$lambda) || !identical(lasso_names, pp$lasso))
    fail("LASSO路径未重现原惩罚参数/变量名单，停止绘图；不要仅修改线的位置。")
  message("LASSO路径核对通过：", length(lasso_names), "项；直接使用原辅助CV结果。")
  message("正在重现最终训练集的Boruta重要性历史，请等待；不进行九模型训练。")
  set.seed(pp$seed)
  bo <- Boruta::Boruta(x = x, y = y, maxRuns = cfg$boruta_max_runs, doTrace = 0,
                       holdHistory = TRUE, num.threads = 2)
  if (any(bo$finalDecision == "Tentative")) bo <- Boruta::TentativeRoughFix(bo)
  boruta_names <- names(bo$finalDecision)[bo$finalDecision == "Confirmed"]
  selected <- names(x)[names(x) %in% intersect(boruta_names, lasso_names)]
  if (!identical(boruta_names, pp$boruta) || !identical(selected, pp$features) ||
      !identical(names(bo$finalDecision), names(pp$boruta_decisions)) ||
      !identical(as.character(bo$finalDecision), as.character(pp$boruta_decisions)))
    fail("Boruta或最终交集没有重现原名单，停止绘图。请检查环境/原始数据，不要换种子凑数量。")
  decision <- read.csv(paths[["features"]], check.names = FALSE, stringsAsFactors = FALSE)
  if (!identical(as.character(decision$Feature), candidate_features) ||
      !identical(as.logical(decision$Boruta), candidate_features %in% boruta_names) ||
      !identical(as.logical(decision$LASSO), candidate_features %in% lasso_names) ||
      !identical(as.logical(decision$Intersection), candidate_features %in% selected))
    fail("最终筛选CSV与重现对象不一致。")
  message("一致性检查全部通过：Boruta ", length(boruta_names), "项，LASSO ",
          length(lasso_names), "项，交集", length(selected), "项；lambda.min；开始绘图。")
  result_norm <- normalizePath(result_dir, winslash = "/", mustWork = TRUE)
  output_norm <- normalizePath(output_dir, winslash = "/", mustWork = FALSE)
  if (tolower(output_norm) %in% tolower(c(result_norm, dirname(normalizePath(raw_file, winslash = "/")))))
    fail("请使用单独的补图子目录，不要直接输出到结果根目录或原始数据目录。")
  if (dir.exists(output_dir) && length(list.files(output_dir, all.files = FALSE)))
    output_dir <- paste0(output_dir, "_", format(Sys.time(), "%Y%m%d_%H%M%S"))
  if (dir.exists(output_dir) && length(list.files(output_dir, all.files = FALSE))) fail("输出目录仍非空，请更换PLOT_OUTPUT_DIR。")
  dir.create(output_dir, recursive = TRUE, showWarnings = FALSE)
  if (!dir.exists(output_dir)) fail("无法建立补图目录：", output_dir)
  output_dir <- normalizePath(output_dir, winslash = "/", mustWork = TRUE)
  beta <- coef_path[rownames(coef_path) != "(Intercept)", , drop = FALSE]
  nnz <- colSums(beta != 0)
  x_values <- if (x_axis == "reference_lambda") log(lambda_grid) else log(cv$Ratio)
  x_label <- if (x_axis == "reference_lambda") "Log_Lambda" else "Log(lambda / lambda_max)"
  path_data <- data.frame(Feature = rep(rownames(beta), times = ncol(beta)),
                          X = rep(x_values, each = nrow(beta)), Coefficient = as.numeric(beta))
  cv$Reference_lambda <- lambda_grid
  cv$Log_reference_lambda <- log(lambda_grid)
  cv$Log_ratio <- log(cv$Ratio)
  cv$Nonzero_coefficients_full_training <- nnz
  cv$X <- x_values
  top_idx <- unique(round(seq(1, nrow(cv), length.out = 10)))
  x_scale <- function() ggplot2::scale_x_continuous(name = x_label,
                                                    sec.axis = ggplot2::dup_axis(name = NULL, breaks = x_values[top_idx], labels = nnz[top_idx]))
  common_theme <- ggplot2::theme_bw(base_size = 10 * text_scale, base_family = "sans") +
    ggplot2::theme(panel.grid = ggplot2::element_blank(),
                   panel.border = ggplot2::element_rect(colour = "black", fill = NA, linewidth = .45),
                   plot.title = ggplot2::element_text(hjust = .5, size = 13 * text_scale, margin = ggplot2::margin(b = 12)),
                   axis.title = ggplot2::element_text(size = 9 * text_scale),
                   axis.text = ggplot2::element_text(size = 8 * text_scale, colour = "#333333"),
                   axis.text.x.top = ggplot2::element_text(size = 6 * text_scale),
                   axis.ticks.x.top = ggplot2::element_blank(),
                   legend.key = ggplot2::element_blank(),
                   plot.margin = ggplot2::margin(28, 24, 25, 25))
  g1 <- ggplot2::ggplot(path_data, ggplot2::aes(x = X, y = Coefficient, colour = Feature, group = Feature)) +
    ggplot2::geom_line(linewidth = .65) + x_scale() +
    ggplot2::labs(title = "Lasso Regression Lambda and Coefficients Plot", y = "Coefficients") +
    common_theme + ggplot2::theme(legend.position = "none", plot.margin = ggplot2::margin(40, 45, 38, 45))
  label_1se <- if (x_axis == "reference_lambda") sprintf("LogLambda_1se=%.3f", x_values[index_1se]) else
    sprintf("LogRatio_1se=%.3f", x_values[index_1se])
  label_min <- if (x_axis == "reference_lambda") sprintf("LogLambda_min=%.3f", x_values[index_min]) else
    sprintf("LogRatio_min=%.3f", x_values[index_min])
  marks <- data.frame(X = x_values[c(index_1se, index_min)],
                      Rule = factor(c(label_1se, label_min), levels = c(label_1se, label_min)))
  g2 <- ggplot2::ggplot(cv, ggplot2::aes(x = X, y = Mean_deviance)) +
    ggplot2::geom_errorbar(ggplot2::aes(ymin = Mean_deviance - SE, ymax = Mean_deviance + SE),
                           width = .055, linewidth = .25, colour = "black") +
    ggplot2::geom_line(colour = "#8EBDA5", linewidth = .4) +
    ggplot2::geom_point(colour = "#8EBDA5", size = .35) +
    ggplot2::geom_vline(data = marks, ggplot2::aes(xintercept = X, colour = Rule),
                        linetype = "dotted", linewidth = .5, show.legend = TRUE) +
    ggplot2::scale_colour_manual(values = setNames(c("blue", "red"), c(label_1se, label_min)), name = NULL) +
    x_scale() + ggplot2::labs(title = "Lasso Regression Lambda and CVM Plot", y = "CVM") +
    common_theme + ggplot2::theme(legend.position = "right", legend.text = ggplot2::element_text(size = 8 * text_scale))
  shadow_names <- c("shadowMax", "shadowMean", "shadowMin")
  show_features <- if (confirmed_only) boruta_names else names(bo$finalDecision)
  history_cols <- c(show_features, shadow_names)
  if (is.null(bo$ImpHistory) || !all(history_cols %in% colnames(bo$ImpHistory))) fail("重现对象没有完整Boruta重要性历史。")
  history <- bo$ImpHistory[, history_cols, drop = FALSE]
  box_data <- data.frame(Variable = rep(colnames(history), each = nrow(history)), Importance = as.numeric(history))
  box_data <- box_data[is.finite(box_data$Importance), , drop = FALSE]
  medians <- tapply(box_data$Importance, box_data$Variable, stats::median)
  box_data$Variable <- factor(box_data$Variable, levels = names(sort(medians)))
  category <- setNames(as.character(bo$finalDecision), names(bo$finalDecision))
  category <- c(category, setNames(shadow_names, shadow_names))
  box_data$Decision <- factor(unname(category[as.character(box_data$Variable)]),
                              levels = c("Confirmed", "Tentative", "Rejected", shadow_names))
  g3 <- ggplot2::ggplot(box_data, ggplot2::aes(x = Variable, y = Importance, fill = Decision)) +
    ggplot2::geom_boxplot(width = .5, linewidth = .3, outlier.colour = "red", outlier.size = .45, outlier.stroke = 0) +
    ggplot2::scale_fill_manual(values = c(Confirmed = "#F8766D", Tentative = "#E6AB02", Rejected = "#999999",
                                          shadowMax = "#7CAE00", shadowMean = "#00BFC4", shadowMin = "#C77CFF"),
                               name = "Final Decision", drop = TRUE) +
    ggplot2::labs(title = "Boruta Feature Filtering Each Variable Importance Box Plot", x = "Variable", y = "Importance") +
    common_theme + ggplot2::theme(axis.text.x = ggplot2::element_text(angle = 45, hjust = 1, size = 7.5 * text_scale),
                                  plot.title = ggplot2::element_text(size = 10.8 * text_scale, margin = ggplot2::margin(b = 5)),
                                  legend.text = ggplot2::element_text(size = 7.5 * text_scale),
                                  legend.title = ggplot2::element_text(size = 8 * text_scale),
                                  legend.key.size = grid::unit(.28, "cm"), plot.margin = ggplot2::margin(16, 6, 14, 6))
  save_plot <- function(p, name, w, h) {
    ggplot2::ggsave(filename = file.path(output_dir, name), plot = p,
                    width = w, height = h, units = "in", device = grDevices::pdf,
                    family = "Helvetica", useDingbats = FALSE, bg = "white", limitsize = FALSE)
  }
  save_plot(g1, "lasso1.pdf", lasso1_width, lasso1_height)
  save_plot(g2, "lasso2.pdf", lasso2_width, lasso2_height)
  save_plot(g3, "boruta.pdf", if (confirmed_only) boruta_width else max(boruta_width, 10), boruta_height)

    invisible(utils::capture.output(
      venn_plot <- VennDiagram::venn.diagram(x = list(boruta = boruta_names, lasso = lasso_names),

                                             scaled = FALSE, disable.logging = TRUE)))
    draw_venn_pdf <- function() {
      grDevices::pdf(file.path(output_dir, "venn.pdf"), width = 8, height = 8, useDingbats = FALSE)
      on.exit(grDevices::dev.off(), add = TRUE)
      grid::grid.newpage(); grid::grid.draw(venn_plot)

  saveRDS(list(lasso_fit = lasso_fit, boruta_object = bo, lasso_cv = cv,
               features = selected, boruta = boruta_names, lasso = lasso_names,

  if (!identical(unname(source_hashes), unname(tools::md5sum(paths))))
    fail("运行期间输入文件发生改变，请不要使用此次输出。")
  audit <- c("PASS: raw MIMIC-IV file hash matches original analysis.",
             if (has_raw_ids) "PASS: saved training patient/stay identities and split membership match." else
               "PASS: saved training source-row identities and split membership match; patient/stay IDs were unavailable.",
             paste("Training records:", nrow(x)), paste("Candidate predictors:", length(candidate_features)),
             paste("Available predictors after training-only missingness/constant checks:", ncol(x)),

             paste("Final predictors:", paste(selected, collapse = ", ")),

             paste("Boruta seed:", pp$seed), paste("Boruta maximum runs:", cfg$boruta_max_runs),
             paste("Boruta history rows:", nrow(bo$ImpHistory)),
             "The original Boruta importance history was not saved; these boxes reconstruct it under matching software/settings and decisions.",
             "No new CV was run. The saved auxiliary-CV means and SEs are plotted; top counts are from the full-training coefficient path.",
             "The CV penalty grid was relative lambda/lambda_max, estimated with fold-specific lambda_max.",
             paste("Plot x-axis:", x_axis),
             "Reference-lambda mode is a coordinate mapping using the full-training lambda_max, not a common absolute lambda across CV folds.",
             "No final nine-algorithm prediction model, classification threshold or performance result was refitted/changed.",
             "PASS: original input files were not modified.")
  writeLines(audit, file.path(output_dir, "Replot_consistency_audit.txt"))
  caption <- c(
    paste0("Figure 3. Predictor selection for the final model refits using the full MIMIC-IV development training set (n = ", nrow(x), ")."),
    paste0("(A) LASSO coefficient paths. (B) Mean binomial deviance with plus/minus one standard error from the saved ", cfg$lasso_folds, "-fold auxiliary cross-validation used for LASSO penalty selection. The red line marks the minimum mean deviance, which was used for selection (lambda.min); the blue line marks the one-standard-error reference, which was not used for selection. Numbers along the upper axes indicate nonzero predictor coefficients in the full-training fit, excluding the intercept; they are not fold-specific selection frequencies."),
    if (x_axis == "reference_lambda") "The LASSO search used lambda/lambda_max with lambda_max estimated in each auxiliary training fold. For consistency with the original display, the horizontal axes show log(lambda) mapped to the full-training lambda_max reference scale; these are not identical absolute penalties across auxiliary folds." else
      "The horizontal axes show log(lambda/lambda_max), where lambda_max was estimated within the relevant training subset.",
    if (confirmed_only) sprintf("(C) Reconstructed Boruta importance distributions for the %d confirmed predictors and the shadow-feature references. Rejected predictors are omitted from this display, and complete decisions are supplied in the accompanying table.", length(boruta_names)) else
      "(C) Reconstructed Boruta importance distributions and decisions for available predictors and shadow-feature references. Variables excluded before Boruta are listed in the accompanying table.",
    sprintf("(D) Overlap between the %d predictors retained by LASSO at lambda.min and the %d predictors confirmed by Boruta, yielding %d predictors for the final model refits.", length(lasso_names), length(boruta_names), length(selected)),
    "These panels illustrate the final training-set selection. Imputation and feature selection were separately repeated within training folds during nested cross-validation; not every fold selected the same predictor set.")
  if (!draw_venn) caption <- caption[!grepl("^\\(D\\)", caption)]
  writeLines(caption, file.path(output_dir, "Figure3_caption_English.txt"))
  message("补图完成：", output_dir)
  message("请检查Replot_consistency_audit.txt中的PASS记录。原模型、阈值和性能结果未改变。")
  invisible(list(output_dir = output_dir, lasso = lasso_names, boruta = boruta_names,
                 features = selected, n_training = nrow(x), input_hashes = source_hashes))
}
